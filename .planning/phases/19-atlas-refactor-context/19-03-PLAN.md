    ```javascript
    'use strict';

    const fs = require('fs');
    const path = require('path');
    const scorer = require('../../../../lib/profile-scorer');
    const claimManager = require('../../../../lib/claim-manager');
    const flowState = require('../../../../lib/flow-state');

    function main() {
      const args = process.argv.slice(2);
      if (args.length < 1) {
        console.error('Usage: node score-and-claim.js <task_json_or_file> [repoRoot]');
        process.exit(1);
      }

      let task;
      try {
        if (fs.existsSync(args[0])) {
          task = JSON.parse(fs.readFileSync(args[0], 'utf8'));
        } else {
          task = JSON.parse(args[0]);
        }
      } catch (err) {
        console.error('Failed to parse task JSON:', err.message);
        process.exit(1);
      }

      const repoRoot = args[1] || process.cwd();

      // 1. Score task risk and select profile
      const scoreResult = scorer.score(task);

      // Determine work mode (default FEATURE, can be overridden)
      const workMode = task.workMode || task.override || 'FEATURE';

      // 2. Claim work unit
      const slug = task.slug || 'task-' + Date.now();
      const owner = task.owner || 'agent';

      let claimSuccess = false;
      try {
        // Mock or invoke claimManager.claim
        // Since claimManager expects options or args, adapt to its API
        // Typically: claim(slug, owner, opts)
        claimSuccess = claimManager.claim(slug, owner, {
          profile: scoreResult.profile,
          write_scope: task.writeScope || []
        });
      } catch (err) {
        console.error('Claim failed:', err.message);
      }

      // 3. Initialize/Load flow state
      let state = flowState.load(repoRoot);
      if (!state) {
        state = {
          schema_version: '2.0',
          run_id: 'run_' + Math.random().toString(36).substr(2, 9),
          feature_slug: slug,
          risk_profile: scoreResult.profile,
          work_mode: workMode,
          stage: 'align',
          status: 'running',
          attempt: 1,
          last_verified_commit: '',
          completed_steps: ['align.score'],
          pending_step: 'align.claim',
          locks: [],
          signals: [],
          consecutive_failures: 0,
          retry_count: 0,
          verified_artifacts: []
        };
      } else {
        state.risk_profile = scoreResult.profile;
        state.work_mode = workMode;
        state.feature_slug = slug;
        if (!state.completed_steps.includes('align.score')) {
          state.completed_steps.push('align.score');
        }
      }

      if (claimSuccess) {
        if (!state.completed_steps.includes('align.claim')) {
          state.completed_steps.push('align.claim');
        }
        state.pending_step = '';
      }

      flowState.save(repoRoot, state);

      const gateResult = {
        stage_id: 'align',
        status: claimSuccess ? 'PASS' : 'FAIL',
        blocking: claimSuccess ? [] : ['Failed to claim work unit ownership'],
        warnings: [],
        artifacts_produced: [path.join('.ai', 'claims', `${slug}.json`), '.ai/state/flow-state.json']
      };

      console.log(JSON.stringify(gateResult, null, 2));
      process.exit(claimSuccess ? 0 : 1);
    }

    if (require.main === module) {
      main();
    }
    ```

    ```javascript
    'use strict';

    const flowState = require('../../../../lib/flow-state');

    function resolveNext(state) {
      const stages = ['align', 'trace', 'lay', 'act', 'settle'];
      const currentIdx = stages.indexOf(state.stage);

      if (state.status === 'done') {
        return { nextStage: null, skipped: [] };
      }

      let nextIdx = currentIdx + 1;
      const skipped = [];

      // Loop to resolve skipped stages/steps based on profile + mode rules
      while (nextIdx < stages.length) {
        const stage = stages[nextIdx];

        if (stage === 'trace') {
          // FAST task scoring skips formal trace spec or uses Trace-Min
          // STANDARD uses Spec-lite; FULL uses Spec-Kit
          // No full stage skips here, just sub-step variants handled in stage trace
        }

        if (stage === 'lay') {
          if (state.work_mode === 'DOCS') {
            // DOCS mode skips test setup in Lay
            skipped.push('lay.test-setup');
          }
        }

        if (stage === 'act') {
          if (state.work_mode === 'DOCS') {
            // DOCS mode skips Act coding iterations entirely
            skipped.push('act');
            nextIdx++;
            continue;
          }
        }

        break;
      }

      const nextStage = nextIdx < stages.length ? stages[nextIdx] : null;

      // Handle align -> trace quick transition for FAST
      if (state.stage === 'align' && state.risk_profile === 'FAST') {
        skipped.push('align-gate');
      }

      return { nextStage, skipped };
    }

    function main() {
      const repoRoot = process.argv[2] || process.cwd();
      const state = flowState.load(repoRoot);
      if (!state) {
        console.error('No flow state found.');
        process.exit(1);
      }

      const { nextStage, skipped } = resolveNext(state);

      if (nextStage) {
        flowState.setStage(state, nextStage);
        flowState.save(repoRoot, state);
      }

      const output = {
        stage_id: state.stage,
        next_stage: nextStage,
        skipped_steps: skipped,
        status: 'PASS',
        blocking: [],
        warnings: [],
        artifacts_produced: ['.ai/state/flow-state.json']
      };

      console.log(JSON.stringify(output, null, 2));
    }

    if (require.main === module) {
      main();
    }

    module.exports = { resolveNext };
    ```
