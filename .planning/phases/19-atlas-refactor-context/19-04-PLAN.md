    ```javascript
    'use strict';

    const fs = require('fs');
    const path = require('path');
    const flowState = require('../../../../lib/flow-state');
    const { resolvePath } = require('../../../../lib/artifact-paths');

    function main() {
      const repoRoot = process.argv[2] || process.cwd();
      const state = flowState.load(repoRoot);

      if (!state) {
        console.error('No flow state found.');
        process.exit(1);
      }

      const blocking = [];
      const warnings = [];

      // 1. Check last_verified_commit
      if (!state.last_verified_commit) {
        blocking.push('Pre-execution base commit hash (last_verified_commit) is not recorded.');
      }

      // 2. Check leases/locks
      // In flow-state v2, locks are stored in the state object under `locks`
      if (!state.locks || state.locks.length === 0) {
        blocking.push('No active file advisory leases acquired in flow state.');
      }

      // 3. Check for failing test file
      // Look for a test file in validators/scripts/ or tests/ that has been created or modified
      // Let's check state.completed_steps for lay.failing-test, or scan modified files
      // As a deterministic fallback, verify that at least one test file exists under tests/ or validators/scripts/
      // or that the state tracks layout validation
      const testDirs = [
        path.join(repoRoot, 'validators', 'scripts'),
        path.join(repoRoot, 'tests')
      ];
      let testExists = false;
      for (const dir of testDirs) {
        if (fs.existsSync(dir)) {
          const files = fs.readdirSync(dir);
          if (files.some(f => f.startsWith('test-') && f.endsWith('.js'))) {
            testExists = true;
            break;
          }
        }
      }

      if (!testExists) {
        blocking.push('No unit or validation test file detected.');
      }

      const status = blocking.length === 0 ? 'PASS' : 'FAIL';
      const gateResult = {
        stage_id: 'lay',
        status: status,
        blocking: blocking,
        warnings: warnings,
        artifacts_produced: []
      };

      console.log(JSON.stringify(gateResult, null, 2));
      process.exit(status === 'PASS' ? 0 : 1);
    }

    if (require.main === module) {
      main();
    }
    ```

  <action>
    Create `.claude/skills/atlas-gates/scripts/act-evaluator.js`:
    ```javascript
    'use strict';

    const flowState = require('../../../../lib/flow-state');

    function main() {
      const repoRoot = process.argv[2] || process.cwd();
      const state = flowState.load(repoRoot);

      if (!state) {
        console.error('No flow state found.');
        process.exit(1);
      }

      const blocking = [];
      const warnings = [];

      // Determine iteration cap based on profile and mode
      let cap = 5; // Default STANDARD
      if (state.risk_profile === 'FAST') {
        cap = 3;
      } else if (state.risk_profile === 'FULL') {
        cap = 8;
      }

      if (['BUGFIX', 'PROTOTYPE'].includes(state.work_mode)) {
        cap = 5;
      }

      if (state.attempt > cap) {
        blocking.push(`Execution attempt count (${state.attempt}) exceeded profile iteration cap (${cap}).`);
      }

      // Stuck detection: check if attempt > 1 and consecutive failures > 1
      if (state.consecutive_failures >= 2) {
        warnings.push(`Stuck execution warning: pending step '${state.pending_step}' failed consecutively.`);
      }

      const status = blocking.length === 0 ? 'PASS' : 'BLOCKED';
      const gateResult = {
        stage_id: 'act',
        status: status,
        blocking: blocking,
        warnings: warnings,
        artifacts_produced: []
      };

      console.log(JSON.stringify(gateResult, null, 2));
      process.exit(status === 'PASS' ? 0 : 1);
    }

    if (require.main === module) {
      main();
    }
    ```

    ```javascript
    const assert = require('assert');
    const fs = require('fs');
    const path = require('path');
    const os = require('os');
    const flowState = require('../../lib/flow-state');

    const tests = [];
    function addTest(name, fn) {
      tests.push({ name, fn });
    }

    let tempDir;

    function setup() {
      tempDir = fs.mkdtempSync(path.join(os.tmpdir(), 'adp-gates-test-'));
      const paths = require('../../lib/artifact-paths');
      const origResolvePath = paths.resolvePath;
      paths.resolvePath = function(dottedKey, variables) {
        if (dottedKey === 'flow_state') {
          return path.join(tempDir, '.ai', 'state', 'flow-state.json');
        }
        return origResolvePath(dottedKey, variables);
      };
    }

    function cleanup() {
      try {
        fs.rmSync(tempDir, { recursive: true, force: true });
      } catch (e) {}
    }

    addTest('act-evaluator respects profile caps', () => {
      const state = {
        schema_version: '2.0',
        run_id: 'run_eval',
        feature_slug: 'eval-feature',
        risk_profile: 'FAST',
        work_mode: 'FEATURE',
        stage: 'act',
        status: 'running',
        attempt: 4, // FAST cap is 3
        completed_steps: [],
        pending_step: 'act.slice-1',
        locks: [],
        signals: [],
        consecutive_failures: 0,
        retry_count: 0,
        verified_artifacts: []
      };

      flowState.save(tempDir, state);

      // Execute code logic directly or shell it
      // Let's shell the script to ensure CLI compliance
      const { execSync } = require('child_process');
      try {
        execSync(`node .claude/skills/atlas-gates/scripts/act-evaluator.js "${tempDir}"`);
        assert.fail('Should have exited with non-zero code');
      } catch (err) {
        const out = JSON.parse(err.stdout.toString());
        assert.strictEqual(out.status, 'BLOCKED');
        assert.ok(out.blocking[0].includes('exceeded profile iteration cap'));
      }

      // STANDARD cap is 5, so attempt=4 should pass
      state.risk_profile = 'STANDARD';
      flowState.save(tempDir, state);
      const res = execSync(`node .claude/skills/atlas-gates/scripts/act-evaluator.js "${tempDir}"`);
      const out = JSON.parse(res.toString());
      assert.strictEqual(out.status, 'PASS');
    });

    // Run tests
    setup();
    let failed = false;
    for (const test of tests) {
      try {
        test.fn();
        console.log(`[PASS] ${test.name}`);
      } catch (err) {
        console.error(`[FAIL] ${test.name}:`, err);
        failed = true;
      }
    }
    cleanup();

    if (failed) {
      process.exit(1);
    }
    ```
