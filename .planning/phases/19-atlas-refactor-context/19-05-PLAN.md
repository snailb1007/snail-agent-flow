    function main() {

      const command = mode === 'validate' ? 'npm run validate' : 'npm test';

      let success = true;

      } catch (err) {

      }

        status: success ? 'PASS' : 'FAIL',
        blocking: success ? [] : [`Settle checks failed running command '${command}'`],
        warnings: [],
        artifacts_produced: []
      };

      console.log(JSON.stringify(gateResult, null, 2));
      process.exit(success ? 0 : 1);
    }

    if (require.main === module) {
      main();
    }

    Create `.claude/skills/atlas-settle/scripts/release-locks.js`:
    ```javascript
    'use strict';

    const path = require('path');
    const flowState = require('../../../../lib/flow-state');
    const claimManager = require('../../../../lib/claim-manager');
    const leaseManager = require('../../../../lib/lease-manager');

    function main() {
      const repoRoot = process.argv[2] || process.cwd();
      const state = flowState.load(repoRoot);

      if (!state) {
        console.error('No flow state found.');
        process.exit(1);
      }

      const featureSlug = state.feature_slug;
      const owner = 'agent';

      // 1. Release active file leases
      let leasesReleased = 0;
      if (state.locks && state.locks.length > 0) {
        for (const lock of state.locks) {
          try {
            // locks store filename or key
            const key = path.basename(lock.file, '.js');
            leaseManager.release(key, owner);
            leasesReleased++;
          } catch (e) {}
        }
        state.locks = [];
      }

      // 2. Release work claims
      let claimReleased = false;
      try {
        claimReleased = claimManager.release(featureSlug, owner);
      } catch (e) {}

      flowState.save(repoRoot, state);

      const gateResult = {
        stage_id: 'settle',
        status: 'PASS',
        blocking: [],
        warnings: [],
        artifacts_produced: []
      };

      console.log(JSON.stringify(gateResult, null, 2));
    }

    if (require.main === module) {
      main();
    }
    ```
